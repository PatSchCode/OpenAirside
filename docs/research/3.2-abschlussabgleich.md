Er verschiebt weder richtige Stationsankunft noch ehrliche Diagnose, FIFO oder nutzbare Engpassverbesserung. Vollständige Check-in-/Kontroll-/Gatevarianten, Räume, Personal, Gepäck und Bauphasen werden durch diesen Abgleich nicht als erledigt ausgegeben.

```rust
//! Diagnostic for 3.2: zero displayed queue must not hide waiting passengers.
//! Requires repository Rust toolchain/dependencies; portable Rust test.
//! Exit0 assertions hold, nonzero diagnostic finding/runtime failure.
use openairside_core::{geometry::MapSize, persons::{PASSENGER_WAITING, PASSENGER_COMPLETE, GROUP_ARRIVING}, world::World};

#[test]
fn zero_aggregate_does_not_hide_waiting_passengers() {
    let pool = rayon::ThreadPoolBuilder::new().num_threads(2).build().unwrap();
    let mut world = World::small_terminal(MapSize::new(64, 64).unwrap(), 2000, 17).unwrap();
    world.rebuild_fields_now(&pool);
    world.build_mut().set_station_throughput(1, 3).unwrap();
    let mut found = None;
    for tick in 0..2000 {
        world.step(&pool, &pool, tick, true);
        let queued: u32 = world.build().stations().iter()
            .map(|s| world.persons().queued_for(s.target)).sum();
        let waiting: Vec<_> = {
            let mut values = Vec::new();
            world.persons().visit(|p| {
                if p.state == PASSENGER_WAITING { values.push((p.x, p.y)); }
            });
            values
        };
        if queued == 0 && !waiting.is_empty() {
            let wait_sum: u64 = world.build().stations().iter()
                .map(|s| world.persons().waiting_ticks_for(s.target)).sum();
            println!("tick={tick}, displayed_queue={queued}, displayed_wait={wait_sum}, actual_waiting={}, first_position={:?}", waiting.len(), waiting[0]);
            found = Some(waiting.len());
            break;
        }
    }
    assert_eq!(found, None, "zero station aggregates conceal actual waiting passengers");
}

#[test]
fn completed_arrivals_reach_the_exit() {
    let pool = rayon::ThreadPoolBuilder::new().num_threads(2).build().unwrap();
    let mut world = World::small_terminal(MapSize::new(64, 64).unwrap(), 2, 17).unwrap();
    world.rebuild_fields_now(&pool);
    let mut arrivals = Vec::new();
    for tick in 0..2000 {
        world.step(&pool, &pool, tick, true);
        world.persons().visit(|p| {

            }
        });

    }
    println!("arrival_end={arrivals:?}; exit_x=44.5, exit_y=31.5");
    assert_eq!(arrivals, vec![(PASSENGER_COMPLETE, 44.5, 31.5)], "arrival must physically reach exit");
```
